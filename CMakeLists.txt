cmake_minimum_required(VERSION 3.16)
project(FortnitePSX CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

include(FetchContent)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
FetchContent_Declare(raylib URL https://github.com/raysan5/raylib/archive/refs/tags/5.5.zip)
FetchContent_MakeAvailable(raylib)

file(GLOB SRC CONFIGURE_DEPENDS src/*.cpp src/*.h)
add_executable(FortnitePSX ${SRC})
target_link_libraries(FortnitePSX raylib)
